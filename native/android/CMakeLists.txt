cmake_minimum_required(VERSION 3.31.6)
project(UrbanChaosAndroidRuntime LANGUAGES C CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
 message(FATAL_ERROR "This runtime requires Android arm64-v8a")
endif()
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SDK "${REPO}/vendor/runtime")
foreach(INPUT URBAN_GENERATED URBAN_GUEST_ARCHIVE URBAN_OVERLAY URBAN_VULKAN_SOURCE URBAN_PROFILE URBAN_SHADER_BUILD)
 if(NOT EXISTS "${${INPUT}}")
  message(FATAL_ERROR "Missing private Android integration input ${INPUT}")
 endif()
endforeach()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/vendor/SDL" SDL EXCLUDE_FROM_ALL)
file(GLOB SDK_SOURCES CONFIGURE_DEPENDS "${SDK}/runtime/*/*.cpp")
list(FILTER SDK_SOURCES EXCLUDE REGEX "/runtime/app/|/core/input\\.cpp$|/gpu/d3d12\\.cpp$|/audio/xaudio2\\.cpp$")
set(RUNTIME_SOURCES)
foreach(SOURCE IN LISTS SDK_SOURCES)
 file(RELATIVE_PATH REL "${SDK}" "${SOURCE}")
 if(EXISTS "${URBAN_OVERLAY}/${REL}")
  list(APPEND RUNTIME_SOURCES "${URBAN_OVERLAY}/${REL}")
 else()
  list(APPEND RUNTIME_SOURCES "${SOURCE}")
 endif()
endforeach()
add_library(main SHARED ${RUNTIME_SOURCES} main.cpp input.cpp bridge.cpp
 "${URBAN_VULKAN_SOURCE}" "${REPO}/native/gpu/vulkan_visibility.cpp"
 "${REPO}/native/game/source.cpp" "${REPO}/native/audio/sdl_output.cpp" "${REPO}/native/diagnostics/android_bridge.cpp")
target_compile_features(main PRIVATE cxx_std_20)
target_compile_options(main PRIVATE -O2 -g0 -fsigned-char -fno-strict-aliasing -include "${CMAKE_CURRENT_LIST_DIR}/platform_compat.h")
target_compile_definitions(main PRIVATE XBE_RAW_VERTICES=1 XBE_NATIVE_COMPONENT_ALIGNMENT=1 XBE_RENDER_THREAD=0)
target_include_directories(main PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${URBAN_OVERLAY}/runtime" "${SDK}/runtime"
 "${SDK}/harness/compat" "${REPO}/native" "${REPO}/host/vulkan-research" "${REPO}/vendor/black-shader/native" "${URBAN_GENERATED}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(main PRIVATE "${URBAN_GUEST_ARCHIVE}" SDL3::SDL3-shared
 "${URBAN_SHADER_BUILD}/liburban_black_shader.a"
 "${URBAN_SHADER_BUILD}/glslang/glslang/libglslang.a"
 "${URBAN_SHADER_BUILD}/glslang/glslang/libglslang-default-resource-limits.a"
 vulkan android log)
target_link_options(main PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
include("${CMAKE_CURRENT_LIST_DIR}/profile.cmake")
