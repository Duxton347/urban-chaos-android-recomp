cmake_minimum_required(VERSION 3.31.6)
project(UrbanResumableTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT EXISTS "${URBAN_CONTEXT_SDK}/tools/recomp/codegen_emit.cpp")
 message(FATAL_ERROR "Explicit patched SDK source copy required")
endif()
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/vendor/zydis" zydis)
add_subdirectory("${REPO}/vendor/tomlplusplus" tomlplusplus)
file(GLOB SOURCES CONFIGURE_DEPENDS "${URBAN_CONTEXT_SDK}/tools/recomp/*.cpp")
add_executable(xbe_recomp_resumable ${SOURCES})
target_include_directories(xbe_recomp_resumable PRIVATE "${URBAN_CONTEXT_SDK}/runtime" "${REPO}/vendor/runtime/runtime")
target_link_libraries(xbe_recomp_resumable PRIVATE Zydis tomlplusplus::tomlplusplus)
