cmake_minimum_required(VERSION 3.31.6)
project(UrbanBlackShaderReuse LANGUAGES C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL ON CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/vendor/glslang" glslang EXCLUDE_FROM_ALL)
if(NOT EXISTS "${URBAN_BLACK_SHADER_SOURCE}/vk_shader.cpp")
 message(FATAL_ERROR "Pinned Black shader compiler source required")
endif()
add_library(urban_black_shader STATIC "${URBAN_BLACK_SHADER_SOURCE}/vk_shader.cpp")
target_compile_features(urban_black_shader PUBLIC cxx_std_20)
target_compile_definitions(urban_black_shader PRIVATE ENABLE_HLSL)
target_include_directories(urban_black_shader PUBLIC "${URBAN_BLACK_SHADER_SOURCE}" PRIVATE "${REPO}/vendor/glslang" "${REPO}/vendor/runtime/runtime" "${REPO}/native")
target_link_libraries(urban_black_shader PUBLIC glslang glslang-default-resource-limits)
if(NOT ANDROID)
add_executable(vulkan_shader_fixture "${REPO}/tests/vulkan_shader_fixture.cpp" "${REPO}/vendor/runtime/runtime/gpu/shaders.cpp" "${REPO}/vendor/runtime/runtime/core/log.cpp")
target_compile_features(vulkan_shader_fixture PRIVATE cxx_std_20)
target_include_directories(vulkan_shader_fixture PRIVATE "${REPO}/vendor/runtime/runtime" "${REPO}/native")
target_link_libraries(vulkan_shader_fixture PRIVATE urban_black_shader)
endif()
