cmake_minimum_required(VERSION 3.31.6)
project(UrbanChaosGeneratedCompileProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(URBAN_GENERATED "" CACHE PATH "Private generated title directory")
if(NOT EXISTS "${URBAN_GENERATED}/x86_func_mapping.cpp" OR NOT EXISTS "${URBAN_GENERATED}/image_symbols.h")
 message(FATAL_ERROR "Real generated sources required; no fallback or stub")
endif()
file(GLOB UNITS CONFIGURE_DEPENDS "${URBAN_GENERATED}/x86_recomp.*.cpp")
if(NOT UNITS)
 message(FATAL_ERROR "No generated code units")
endif()
add_library(urban_generated_compile_probe STATIC ${UNITS} "${URBAN_GENERATED}/x86_func_mapping.cpp")
target_include_directories(urban_generated_compile_probe PRIVATE "${URBAN_GENERATED}" "${URBAN_RUNTIME_HEADERS}" "${CMAKE_CURRENT_LIST_DIR}/../../vendor/runtime/runtime")
set(URBAN_GUEST_OPTIMIZATION "0" CACHE STRING "Guest optimization: 0 for compile probes, 1 or 2 for runtime experiments")
set_property(CACHE URBAN_GUEST_OPTIMIZATION PROPERTY STRINGS 0 1 2)
if(NOT URBAN_GUEST_OPTIMIZATION MATCHES "^[012]$")
 message(FATAL_ERROR "Guest optimization must be 0, 1 or 2; fast-math is not supported")
endif()
target_compile_options(urban_generated_compile_probe PRIVATE "-O${URBAN_GUEST_OPTIMIZATION}" -g0 -fsigned-char -fno-strict-aliasing)
# A compile probe only: this archive is not linked to a game host and proves no boot.
