cmake_minimum_required(VERSION 3.31.6)
project(UrbanVulkanReports LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT EXISTS "${URBAN_VULKAN_HEADERS}/vulkan/vulkan.h")
 message(FATAL_ERROR "Fixed Vulkan headers are required")
endif()
find_library(URBAN_VULKAN_LIBRARY NAMES vulkan libvulkan.so.1 REQUIRED)
add_library(urban_vulkan_visibility STATIC "${REPO}/native/gpu/vulkan_visibility.cpp")
target_compile_features(urban_vulkan_visibility PUBLIC cxx_std_20)
target_include_directories(urban_vulkan_visibility PUBLIC "${REPO}/native" "${URBAN_VULKAN_HEADERS}")
target_link_libraries(urban_vulkan_visibility PUBLIC "${URBAN_VULKAN_LIBRARY}")
add_executable(vulkan_reports_fixture "${REPO}/tests/vulkan_reports_fixture.cpp")
target_compile_features(vulkan_reports_fixture PRIVATE cxx_std_20)
target_link_libraries(vulkan_reports_fixture PRIVATE urban_vulkan_visibility)
