cmake_minimum_required(VERSION 3.31.6)
project(UrbanChaosHostTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/vendor/zydis zydis)
add_subdirectory(${ROOT}/vendor/tomlplusplus tomlplusplus)
file(GLOB RECOMP_SOURCES "${ROOT}/vendor/runtime/tools/recomp/*.cpp")
add_executable(xbe_recomp ${RECOMP_SOURCES})
add_executable(xbe_symbols ${ROOT}/vendor/runtime/tools/symbols/main.cpp)
foreach(t IN ITEMS xbe_recomp xbe_symbols)
 target_include_directories(${t} PRIVATE ${ROOT}/vendor/runtime/runtime)
 target_link_libraries(${t} PRIVATE Zydis tomlplusplus::tomlplusplus)
endforeach()
