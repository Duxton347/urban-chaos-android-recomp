cmake_minimum_required(VERSION 3.31.6)
project(UrbanGameSource LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SDK "${REPO}/vendor/runtime")
if(NOT EXISTS "${URBAN_OVERLAY}/runtime/kernel/disc.cpp")
 message(FATAL_ERROR "Bounded disc overlay required")
endif()
add_executable(game_source_probe main.cpp "${REPO}/native/game/source.cpp"
 "${URBAN_OVERLAY}/runtime/kernel/disc.cpp" "${SDK}/runtime/kernel/crypto.cpp")
target_include_directories(game_source_probe PRIVATE "${URBAN_OVERLAY}/runtime" "${SDK}/runtime" "${REPO}/native")
target_compile_options(game_source_probe PRIVATE -O2 -UNDEBUG -fsigned-char -fno-strict-aliasing)
target_link_libraries(game_source_probe PRIVATE pthread)
