cmake_minimum_required(VERSION 3.31.6)
project(UrbanChaosRuntimeCoreAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK "${CMAKE_CURRENT_LIST_DIR}/../../vendor/runtime")
file(GLOB KERNEL_SOURCES CONFIGURE_DEPENDS "${SDK}/runtime/kernel/*.cpp")
add_library(xbox_kernel_core STATIC ${KERNEL_SOURCES} "${SDK}/runtime/core/log.cpp" "${SDK}/runtime/cpu/dispatch.cpp")
target_include_directories(xbox_kernel_core PUBLIC "${SDK}/runtime" PRIVATE "${SDK}/harness/compat")
target_compile_options(xbox_kernel_core PRIVATE -fsigned-char -fno-strict-aliasing)
add_executable(test_runtime_memory test_runtime_memory.cpp)
target_link_libraries(test_runtime_memory PRIVATE xbox_kernel_core pthread)
enable_testing()
add_test(NAME runtime_memory_aliases COMMAND test_runtime_memory)
