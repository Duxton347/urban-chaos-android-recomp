# Compile the actual patched SDK getter bodies against entirely authored metadata and resource state.
set(SURFACE_FIXTURE "${CMAKE_CURRENT_BINARY_DIR}/surface-fixture")

file(WRITE "${SURFACE_FIXTURE}/image_symbols.h" "#pragma once\n#define SYM_D3DDevice_GetBackBuffer2 00012000\n#define SYM_D3DDevice_GetRenderTarget2 00012010\n#define SYM_D3DDevice_GetDepthStencilSurface2 00012020\n#define SYM_D3D_PublicAddRef 00013000\n#define HAS_D3D_PublicAddRef 1\n")
foreach(VARIANT patched legacy)
 if(VARIANT STREQUAL "patched")
  set(SOURCE "${URBAN_OVERLAY}/runtime/xdk/d3d8.cpp")
  set(MARKER "// Returned views require the retail public retain operation")
  set(FRAGMENT "surface_getters.inc")
 else()
  set(SOURCE "${SDK}/runtime/xdk/d3d8.cpp")
  set(MARKER "HLE_STDCALL(D3DDevice_GetBackBuffer2, 1) {")
  set(FRAGMENT "legacy_surface_getters.inc")
 endif()
 file(READ "${SOURCE}" GETTERS)
 string(FIND "${GETTERS}" "${MARKER}" START)
 string(FIND "${GETTERS}" "\nnamespace {\nstd::atomic<uint32_t> g_presents" END)
 if(START LESS 0 OR END LESS START)
  message(FATAL_ERROR "SDK surface getter boundaries changed; review the fixture extraction")
 endif()
 math(EXPR LENGTH "${END}-${START}")
 string(SUBSTRING "${GETTERS}" ${START} ${LENGTH} GETTERS)
 file(WRITE "${SURFACE_FIXTURE}/${FRAGMENT}" "${GETTERS}")
endforeach()
foreach(TARGET surface_reference_probe legacy_surface_reference_probe)
 add_executable(${TARGET} surface_references.cpp "${SDK}/runtime/cpu/dispatch.cpp"

 target_compile_options(${TARGET} PRIVATE -O1 -UNDEBUG -fsigned-char -fno-strict-aliasing)
 target_link_libraries(${TARGET} PRIVATE pthread)
endforeach()
