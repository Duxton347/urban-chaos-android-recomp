cmake_minimum_required(VERSION 3.31.6)
project(UrbanChaosResearchRuntime LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
 message(FATAL_ERROR "This research host currently supports Linux x86_64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SDK "${REPO}/vendor/runtime")
foreach(INPUT URBAN_GENERATED URBAN_GUEST_ARCHIVE URBAN_OVERLAY URBAN_PROFILE)
 if(NOT EXISTS "${${INPUT}}")
  message(FATAL_ERROR "Missing private research input ${INPUT}")
 endif()
endforeach()
file(GLOB SDK_SOURCES CONFIGURE_DEPENDS "${SDK}/runtime/*/*.cpp")
list(FILTER SDK_SOURCES EXCLUDE REGEX "/runtime/app/|/core/input\\.cpp$|/gpu/d3d12\\.cpp$|/audio/xaudio2\\.cpp$")
set(RUNTIME_SOURCES)
foreach(SOURCE IN LISTS SDK_SOURCES)
 file(RELATIVE_PATH REL "${SDK}" "${SOURCE}")
 if(EXISTS "${URBAN_OVERLAY}/${REL}")
  list(APPEND RUNTIME_SOURCES "${URBAN_OVERLAY}/${REL}")
 else()
  list(APPEND RUNTIME_SOURCES "${SOURCE}")
 endif()
endforeach()
file(GLOB HARNESS_SOURCES CONFIGURE_DEPENDS "${SDK}/harness/*.cpp")
list(FILTER HARNESS_SOURCES EXCLUDE REGEX "/null_audio\\.cpp$")
set(PATCHED_HARNESS)
foreach(SOURCE IN LISTS HARNESS_SOURCES)
 file(RELATIVE_PATH REL "${SDK}" "${SOURCE}")
 if(EXISTS "${URBAN_OVERLAY}/${REL}")
  list(APPEND PATCHED_HARNESS "${URBAN_OVERLAY}/${REL}")
 else()
  list(APPEND PATCHED_HARNESS "${SOURCE}")
 endif()
endforeach()
add_executable(urban_research ${RUNTIME_SOURCES} ${PATCHED_HARNESS} game_info.cpp audio_sink.cpp probes.cpp "${REPO}/host/runtime-probes/gpu_probes.cpp" "${REPO}/host/audio-device/test_device_lifetime.cpp")
target_include_directories(urban_research PRIVATE "${URBAN_OVERLAY}/runtime" "${SDK}/runtime" "${URBAN_OVERLAY}/harness" "${SDK}/harness" "${SDK}/harness/compat" "${URBAN_GENERATED}" "${REPO}/native")
target_compile_definitions(urban_research PRIVATE XBE_RAW_VERTICES=0 XBE_RENDER_THREAD=0)
target_compile_options(urban_research PRIVATE -O1 -g1 -UNDEBUG -march=x86-64-v2 -fsigned-char -fno-strict-aliasing)
target_link_options(urban_research PRIVATE -rdynamic)
target_link_libraries(urban_research PRIVATE "${URBAN_GUEST_ARCHIVE}" pthread)
file(READ "${URBAN_PROFILE}" PROFILE)
string(JSON COUNT LENGTH "${PROFILE}" bindings)
math(EXPR LAST "${COUNT}-1")
foreach(I RANGE ${LAST})
 string(JSON NAME MEMBER "${PROFILE}" bindings ${I})
 if(NOT NAME MATCHES "^(IDirectSound_AddRef|IDirectSoundBuffer_GetVoiceProperties|IDirectSoundBuffer_SetPitch|IDirectSoundBuffer_SetMode|IDirectSound_CreateSoundStream|CDirectSoundStream_SetOutputBuffer|CDirectSoundStream_SetPitch|D3D_FvfShader|D3DDevice_IsBusy|D3DDevice_RunPushBuffer|D3DDevice_GetVisibilityTestResult|D3D_PublicAddRef|IDirectSoundBuffer_SetConeAngles|IDirectSoundBuffer_SetConeOrientation|IDirectSoundBuffer_SetConeOutsideVolume|IDirectSoundBuffer_SetDistanceFactor|IDirectSoundBuffer_SetRolloffFactor|IDirectSoundBuffer_SetDopplerFactor)$")
  message(FATAL_ERROR "Unreviewed research binding ${NAME}")
 endif()
 string(JSON ADDRESS GET "${PROFILE}" bindings ${NAME} address)
 string(LENGTH "${ADDRESS}" LENGTH)
 if(NOT LENGTH EQUAL 8 OR NOT ADDRESS MATCHES "^[0-9a-fA-F]+$")
  message(FATAL_ERROR "Invalid binding address")
 endif()
 target_compile_definitions(urban_research PRIVATE "SYM_${NAME}=${ADDRESS}" "HAS_${NAME}=1")
endforeach()
# This host drops audio output and uses the SDK reference/test renderer. No playability claim.
set(URBAN_OBSERVER "" CACHE FILEPATH "Optional private forwarding observer for crash diagnosis")
if(URBAN_OBSERVER)
 if(NOT EXISTS "${URBAN_OBSERVER}")
  message(FATAL_ERROR "Research observer missing")
 endif()
 target_sources(urban_research PRIVATE "${URBAN_OBSERVER}")
endif()

string(JSON RESUMABLE ERROR_VARIABLE RESUME_ERROR GET "${PROFILE}" resumable_guest)
if(RESUME_ERROR STREQUAL "NOTFOUND" AND RESUMABLE)
 target_compile_definitions(urban_research PRIVATE XBE_RESUMABLE_GUEST=1)
endif()
